Papa.parse(csvURL, { download: true, header: false, skipEmptyLines: true, complete: function(results) { const rawData = results.data.slice(1); // ميزة "الذكاء الاصطناعي" لدمج التكرار: const cleanData = []; const seenNames = new Set(); rawData.forEach(row => { const name = row[0]?.trim(); const parent = row[1]?.trim(); if (name && name !== "") { // إذا تكرر الاسم كـ (ابن)، نأخذه مرة واحدة فقط لمنع تعطل الشجرة if (!seenNames.has(name)) { cleanData.push({ name: name, parent: parent }); seenNames.add(name); } } }); // التأكد من أن كل "أب" موجود في قائمة "الأسماء" لضمان ظهور الشجرة const namesList = new Set(cleanData.map(d => d.name)); cleanData.forEach(d => { if (d.parent && d.parent !== "" && !namesList.has(d.parent)) { // إضافة الأب المفقود تلقائياً كجذع للشجرة cleanData.push({ name: d.parent, parent: "" }); namesList.add(d.parent); } }); initTree(cleanData); } });
