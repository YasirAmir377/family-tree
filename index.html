Papa.parse(csvURL, { download: true, header: false, skipEmptyLines: true, complete: function(results) { const allRows = results.data; const relationsMap = new Map(); const allNames = new Set(); // تصحيح اتجاه القراءة: العمود الحالي (الابن) يُربط بالعمود الذي يليه (الأب) allRows.forEach(row => { for (let i = 0; i < row.length - 1; i++) { let child = row[i]?.trim(); let parent = row[i+1]?.trim(); if (child && parent && child !== "" && parent !== "") { allNames.add(child); allNames.add(parent); relationsMap.set(child, parent); } } }); // تجهيز البيانات وتوحيد الجذور (لجعل الجد الأكبر في الأعلى) const cleanData = []; const rootName = "الأصل_المخفي"; cleanData.push({ name: rootName, parent: "" }); allNames.forEach(name => { let parent = relationsMap.get(name); // إذا لم يكن للشخص أب في الجدول، نربطه بالأصل ليصبح في أعلى الشجرة if (!parent || !allNames.has(parent)) { parent = rootName; } cleanData.push({ name: name, parent: parent }); }); drawD3Tree(cleanData, rootName); } }); function drawD3Tree(data, rootName) { try { const width = window.innerWidth; const height = window.innerHeight; svg = d3.select("#tree-container").append("svg").attr("width", width).attr("height", height); g = svg.append("g"); zoom = d3.zoom().scaleExtent([0.1, 3]).on("zoom", (e) => g.attr("transform", e.transform)); svg.call(zoom); const root = d3.stratify().id(d => d.name).parentId(d => d.parent)(data); const treeLayout = d3.tree().nodeSize([160, 200]); treeLayout(root); g.selectAll(".link").data(root.links()).enter().append("path") .attr("class", "link") .attr("d", d3.linkVertical().x(d => d.x).y(d => d.y)) .style("opacity", d => d.source.id === rootName ? 0 : 1); const node = g.selectAll(".node").data(root.descendants()).enter().append("g") .attr("class", "node") .attr("transform", d => `translate(${d.x},${d.y})`) .style("opacity", d => d.id === rootName ? 0 : 1); node.append("rect").attr("x", -70).attr("y", -20).attr("width", 140).attr("height", 40); node.append("text").attr("dy", 5).text(d => d.id); resetZoom(); } catch (error) { console.error(error); document.getElementById("errorBox").style.display = "block"; document.getElementById("errorBox").innerText = "يوجد دوران في الأسماء! يرجى مراجعة الإكسل والتأكد أن الابن ليس أباً لجدّه."; } } function resetZoom() { svg.transition().duration(750).call(zoom.transform, d3.zoomIdentity.translate(window.innerWidth/2, 50).scale(0.8)); } function exportPDF() { const element = document.getElementById('tree-container'); html2pdf().from(element).set({ margin: 10, filename: 'Family_Tree.pdf', jsPDF: { unit: 'mm', format: 'a2', orientation: 'landscape' } }).save(); }
